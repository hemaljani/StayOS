# Chat Agent Service - Python dependencies
# AgentCore Runtime container with BedrockAgentCoreApp
# bridging browser WebSocket connections to a per-session Strands Agent that
# calls hotel operations tools via the AgentCore Gateway (MCP over SigV4).

# AgentCore SDK (includes Starlette, Uvicorn, websockets)
# Provides BedrockAgentCoreApp framework, /ping health check, and /ws WebSocket routing
bedrock-agentcore>=1.18.0

# Strands Agents SDK - model-driven agent framework providing the Agent class,
# BedrockModel (Claude Sonnet via the Converse API), and MCPClient
# (strands.tools.mcp.mcp_client) used to discover and invoke Gateway tools.
# NOTE: `strands-agents-tools-mcp` does not exist on PyPI - MCP client support
# ships as part of the core `strands-agents` package (strands.tools.mcp), so no
# separate MCP package is required here.
strands-agents>=1.0.0,<2.0.0

# SigV4-signed MCP client transport for connecting to AgentCore Gateway over
# Streamable HTTP. Provides aws_iam_streamablehttp_client used by server.py to
# open the Gateway MCP connection with IAM auth (no OAuth needed).
mcp-proxy-for-aws>=1.6.0,<2.0.0

# AWS SDK for Cognito identity resolution (GetUser) in identity_resolver.py
boto3>=1.34.0
botocore>=1.34.0

# Structured logging (Powertools Logger works outside Lambda context)
# Provides structured JSON logging with context enrichment for CloudWatch Logs
aws-lambda-powertools>=2.30.0

# HTTP client for IMDS credential fetching (AgentCore provides credentials via IMDS)
# Used by the startup credential loader and background refresh task
requests>=2.31.0
