# PULSE Triage Agent service - Python dependencies
# Request-response AgentCore Runtime container: a Strands Agent backed by Claude
# Sonnet that gathers hotel-ops facts via the shared StayOS AgentCore Gateway
# (MCP over SigV4), generates a validated triageBrief reusing the pulse.triage
# package, and attaches it to pulse-alerts + publishes ALERT_UPDATED.

# AgentCore SDK (includes Starlette, Uvicorn). Provides BedrockAgentCoreApp,
# the /ping health check, and the /invocations request-response entrypoint.
bedrock-agentcore>=1.18.0

# Strands Agents SDK - the Agent class, BedrockModel (Claude Sonnet via the
# Converse API), and MCPClient (strands.tools.mcp.mcp_client) used to discover
# and invoke the shared Gateway's tools. MCP client support ships inside the
# core strands-agents package (no separate MCP package needed).
strands-agents>=1.0.0,<2.0.0

# SigV4-signed MCP client transport for connecting to the AgentCore Gateway over
# Streamable HTTP with IAM auth (aws_iam_streamablehttp_client) - the same
# transport the LUMI chat agent uses.
mcp-proxy-for-aws>=1.6.0,<2.0.0

# AWS SDK - bundled into the image (not relying on the runtime's version) for
# the Bedrock Converse fallback, the conditional pulse-alerts attach (DynamoDB),
# and the SigV4-signed AppSync Events publish (PYQUALITY-06).
boto3>=1.35.0,<2.0.0
botocore>=1.35.0,<2.0.0

# Structured JSON logging and best-effort EMF metrics (Powertools works outside
# Lambda). Used by the service and by the reused pulse.common/pulse.triage code.
aws-lambda-powertools>=2.40.0,<3.0.0

# HTTP client for IMDS credential fetching on AgentCore Runtime (matches the
# LUMI chat agent).
requests>=2.31.0

# NOTE: the `pulse` package (pulse.triage.*, pulse.common.*, pulse.delivery.*)
# is NOT a PyPI dependency - it is copied into the image from the backend
# src/ layout by the container build (see Containerfile). It is bundled, not
# pip-installed from an index.
